{{/* Responsive with a WebP source and the original as fallback. Params: src — path under /static, e.g. "images/cinema/intro-bayglass.jpg" alt — alt text (pass "" only for genuinely decorative images) class — classes for the sizes — the sizes attribute (optional) priority — true for the LCP image: eager + fetchpriority=high Reads data/imagedims.json for width/height (prevents layout shift). */}} {{ $src := .src }} {{ $file := path.Base $src }} {{ $webp := replaceRE `\.(jpe?g|png|JPG)$` ".webp" $src }} {{ $dims := index hugo.Data.imagedims $file }} {{ $w := "" }}{{ $h := "" }} {{ with $dims }}{{ $w = .w }}{{ $h = .h }}{{ end }} {{/* Phone/tablet copies (name-640w.webp, name-1080w.webp) are listed when they exist, so small screens download a small file. */}} {{ $srcset := slice }} {{ range $size := slice 640 1080 }} {{ $variant := replaceRE `\.webp$` (printf "-%dw.webp" $size) $webp }} {{ if fileExists (printf "static/%s" $variant) }} {{ $srcset = $srcset | append (printf "%s %dw" ($variant | relURL) $size) }} {{ end }} {{ end }} {{ if $srcset }} {{ $srcset = $srcset | append (printf "%s %vw" ($webp | relURL) ($w | default 1920)) }} {{ else }} {{ $srcset = slice ($webp | relURL) }} {{ end }}
